\section{Probl\'ematique}

\begin{frame}{Contexte : R\'eseaux V\'ehiculaires (VANET)}
\begin{columns}
\column{0.5\textwidth}
\textbf{D\'efinition :}
\begin{itemize}
    \item R\'eseau ad-hoc form\'e par v\'ehicules communicants
    \item Communications V2V, V2I, V2N, V2P
    \item Mobilit\'e \'elev\'ee (50-150 km/h)
    \item Topologie dynamique
\end{itemize}

\column{0.5\textwidth}
\textbf{Applications critiques :}
\begin{itemize}
    \item \'Evitement de collision
    \item Gestion intelligente du trafic
    \item Conduite autonome
    \item Services d'urgence
\end{itemize}
\end{columns}

\vspace{0.3cm}
\begin{alertblock}{Exigences de latence}
S\'ecurit\'e routi\`ere : $<$ 20 ms | Infotainment : $<$ 100 ms | Fiabilit\'e : 99.999\%
\end{alertblock}
\end{frame}

\begin{frame}{D\'efis et Limites du Cloud Computing}
\begin{columns}
\column{0.5\textwidth}
\textbf{D\'efis des VANET :}
\begin{itemize}
    \item \textcolor{fogred}{Congestion r\'eseau} : Pics de trafic
    \item \textcolor{fogred}{Latence \'elev\'ee} : 100-200 ms (Cloud)
    \item \textcolor{fogred}{Surcharge Cloud} : Centralisation
    \item Bande passante limit\'ee
    \item Mobilit\'e des n\oe uds
\end{itemize}

\column{0.5\textwidth}
\textbf{Limites du Cloud seul :}
\begin{itemize}
    \item[$\times$] RTT trop \'elev\'e ($\sim$125 ms)
    \item[$\times$] Point unique de d\'efaillance
    \item[$\times$] Co\^uts de bande passante
    \item[$\times$] Pas de conscience locale
    \item[$\times$] Inadapt\'e au temps r\'eel
\end{itemize}
\end{columns}

\vspace{0.5cm}
\begin{block}{Constat}
Le Cloud seul \textbf{ne peut pas} r\'epondre aux exigences des applications v\'ehiculaires temps r\'eel.
\end{block}
\end{frame}

\section{M\'ethodologie Adopt\'ee}

\begin{frame}{Sc\'enarios de Simulation}
\textbf{3 sc\'enarios comparatifs :}

\begin{table}
\centering
\begin{tabular}{lccc}
\toprule
\textbf{Sc\'enario} & \textbf{Fog Nodes} & \textbf{SCOOT} & \textbf{Description} \\
\midrule
Edge + Cloud & 0\% & D\'esactiv\'e & R\'ef\'erence (sans fog) \\
Full Fog - Low & $\sim$15\% & Actif & Peu de fog nodes \\
Full Fog - High & $\sim$50\% & Actif & Beaucoup de fog nodes \\
\bottomrule
\end{tabular}
\end{table}

\vspace{0.3cm}
\textbf{Architecture 3-tiers :}
\begin{itemize}
    \item \textbf{Edge} : V\'ehicules (traitement local, 500 MIPS)
    \item \textbf{Fog} : RSU + V\'ehicules fog-capable (1500-5000 MIPS)
    \item \textbf{Cloud} : Serveur distant (20000 MIPS, 125 ms RTT)
\end{itemize}
\end{frame}

\begin{frame}{Strat\'egie d'Offloading}
\textbf{Mod\`ele math\'ematique de d\'ecision :}

\begin{equation}
\text{destination} = \arg\min_{d \in \{local, fog, cloud\}} \left( T_d + \alpha \cdot L_d + \beta \cdot E_d \right)
\end{equation}

O\`u : $T_d$ = temps d'ex\'ecution, $L_d$ = latence, $E_d$ = co\^ut \'energ\'etique

\vspace{0.3cm}
\textbf{Temps total d'offloading :}
\begin{align}
T_{offload} &= T_{upload} + T_{exec} + T_{download} \\
T_{upload} &= \frac{D_{input}}{BW_{up}}, \quad T_{exec} = \frac{I}{MIPS}, \quad T_{download} = \frac{D_{output}}{BW_{down}}
\end{align}

\textbf{Coefficients :} $\alpha = 0.4$ (latence), $\beta = 0.3$ (\'energie), $\gamma = 0.3$ (temps)
\end{frame}

\begin{frame}{Outils et Algorithmes Utilis\'es}
\begin{table}
\centering
\begin{tabular}{lp{8cm}}
\toprule
\textbf{Outil} & \textbf{R\^ole} \\
\midrule
\textbf{SUMO} & Simulateur de trafic routier (TraCI API) \\
\textbf{iFogSim} & Simulateur Fog Computing (Java, calcul m\'etriques) \\
\textbf{Python/Flask} & Orchestrateur + Dashboard temps r\'eel \\
\textbf{SCOOT} & Algorithme adaptatif de contr\^ole des feux \\
\bottomrule
\end{tabular}
\end{table}

\vspace{0.3cm}
\textbf{Algorithme SCOOT (Split Cycle Offset Optimization Technique) :}
\begin{itemize}
    \item Ajustement dynamique des phases de feux
    \item Bas\'e sur les files d'attente en temps r\'eel
    \item Optimisation par zone RSU
    \item D\'ecisions : split (dur\'ee phases), cycle, offset
\end{itemize}
\end{frame}

\begin{frame}[fragile]{R\'esultats : Exemples de D\'ecisions SCOOT}
\textbf{D\'ecisions SCOOT observ\'ees (High Fog) :}

\begin{lstlisting}[language=bash]
[SCOOT SPLIT] J1: EW=13s, NS=37s (PI=0.10, NS queue=63m, EW queue=0m)
[EXEC FOG->SUMO] J1: ALTERNANCE FORCEE vers E-W (congestion critique)
[EXEC FOG->SUMO] RSU_J1: 29 vehicules REROUTES
[EXEC CRITIQUE] RSU_J1: URGENCE - 29 vehicules reroutes, feu bascule!
\end{lstlisting}

\vspace{0.3cm}
\textbf{Interpr\'etation :}
\begin{itemize}
    \item \textbf{PI} (Performance Index) : Indicateur de congestion (0 = fluide, 1 = bloqu\'e)
    \item \textbf{Split} : R\'epartition du temps de vert (NS vs EW)
    \item \textbf{Queue} : Longueur des files d'attente en m\`etres
    \item D\'ecisions en \textbf{temps r\'eel} ($<$ 50 ms)
\end{itemize}
\end{frame}

\section{Discussion}

\begin{frame}{Discussion des R\'esultats}
\textbf{Observations principales :}

\begin{enumerate}
    \item \textbf{R\'eduction de latence significative} : 84\% avec High Fog
    \begin{itemize}
        \item Traitement local \'evite le RTT cloud (125 ms $\rightarrow$ 20 ms)
        \item Plus de fog nodes = meilleure r\'epartition de charge
    \end{itemize}
    
    \item \textbf{SCOOT plus efficace avec plus de fog nodes}
    \begin{itemize}
        \item High Fog : 360 ajustements vs Low Fog : 53 ajustements
        \item D\'ecisions plus fr\'equentes gr\^ace \`a la latence r\'eduite
    \end{itemize}
    
    \item \textbf{Gestion proactive de la congestion}
    \begin{itemize}
        \item 2077 reroutages en High Fog (vs 0 en Edge+Cloud)
        \item Anticipe les embouteillages au lieu de les subir
    \end{itemize}
\end{enumerate}

\begin{alertblock}{Limite observ\'ee}
High Fog montre 90\% de congestion car le d\'ebit v\'ehiculaire est plus \'elev\'e. Mais cette congestion est \textbf{g\'er\'ee activement} contrairement \`a Edge+Cloud.
\end{alertblock}
\end{frame}

\begin{frame}{Avantages et Limites}
\begin{columns}
\column{0.5\textwidth}
\textbf{Avantages du Fog V\'ehiculaire :}
\begin{itemize}
    \item[$\checkmark$] Latence $<$ 25 ms
    \item[$\checkmark$] D\'ecisions temps r\'eel
    \item[$\checkmark$] Pas de point unique de d\'efaillance
    \item[$\checkmark$] \'Economie bande passante cloud
    \item[$\checkmark$] Conscience du contexte local
\end{itemize}

\column{0.5\textwidth}
\textbf{Limites identifi\'ees :}
\begin{itemize}
    \item[$\times$] D\'epend de la densit\'e fog
    \item[$\times$] Mobilit\'e = instabilit\'e
    \item[$\times$] H\'et\'erog\'en\'eit\'e des ressources
    \item[$\times$] Infrastructure RSU n\'ecessaire
    \item[$\times$] S\'ecurit\'e complexe
\end{itemize}
\end{columns}
\end{frame}

\section{Conclusion et Perspectives}

\begin{frame}{Conclusion}
\textbf{Ce que nous avons d\'emontr\'e :}

\begin{enumerate}
    \item Le \textbf{Fog Computing} r\'eduit la latence de \textbf{84\%} (125 $\rightarrow$ 20 ms)
